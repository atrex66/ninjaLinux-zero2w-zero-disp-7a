function(rt_module name source)
  add_library(${name} MODULE ${source})
  target_compile_definitions(${name} PRIVATE RTAPI SIM __MODULE__)
  target_link_options(${name} PRIVATE -Wl,-Bsymbolic)
  target_link_libraries(${name} PRIVATE m)
  set_target_properties(${name} PROPERTIES PREFIX "")
  install(TARGETS ${name} LIBRARY DESTINATION rtlib COMPONENT runtime)
endfunction()
rt_module(hal_lib ${U}/src/hal/hal_lib.c)
foreach(name threads debounce pid watchdog encoder counter siggen pwmgen stepgen mux_generic
    supply weighted_sum sampler streamer)
  rt_module(${name} ${U}/src/hal/components/${name}.c)
endforeach()
foreach(name sampler streamer)
  add_executable(hal${name} ${U}/src/hal/components/${name}_usr.c)
  target_compile_definitions(hal${name} PRIVATE ULAPI)
  target_link_libraries(hal${name} PRIVATE automationhal)
  install(TARGETS hal${name} RUNTIME DESTINATION bin COMPONENT runtime)
endforeach()

  COMMAND ${Python3_EXECUTABLE} -c "from yapps.cli_tool import generate; import sys; sys.exit(generate(sys.argv[1], sys.argv[2]))"
    ${U}/src/hal/utils/halcompile.g ${HALCOMPILE}
  DEPENDS ${U}/src/hal/utils/halcompile.g VERBATIM)
add_custom_target(halcompile_parser ALL DEPENDS ${HALCOMPILE})
file(GLOB COMP_SOURCES CONFIGURE_DEPENDS ${U}/src/hal/components/*.comp)
foreach(comp ${COMP_SOURCES})
  get_filename_component(name ${comp} NAME_WE)
  set(generated ${CMAKE_CURRENT_BINARY_DIR}/components/${name}.c)
  add_custom_command(OUTPUT ${generated}
    COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/components
    COMMAND ${Python3_EXECUTABLE} ${HALCOMPILE} --preprocess -o ${generated} ${comp}
    DEPENDS ${HALCOMPILE} ${comp} VERBATIM)
  rt_module(${name} ${generated})
endforeach()
